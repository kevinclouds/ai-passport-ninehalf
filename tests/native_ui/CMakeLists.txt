cmake_minimum_required(VERSION 3.16)
project(nt_ui C CXX ASM)
set(LV_BUILD_CONF_PATH ${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
get_filename_component(NT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${NT_ROOT}/managed_components/lvgl__lvgl" lvgl)
add_executable(nt_ui test.c "${NT_ROOT}/main/ninehalf_ui.c" "${NT_ROOT}/main/ninehalf_model.c" "${NT_ROOT}/main/ninehalf_fonts.c" "${NT_ROOT}/main/ninehalf_stream.c" "$ENV{IDF_PATH}/components/json/cJSON/cJSON.c")
target_include_directories(nt_ui PRIVATE "$ENV{IDF_PATH}/components/json/cJSON" stubs "${NT_ROOT}/main" "${NT_ROOT}/tests/bsp_stubs" "${NT_ROOT}/components/bsp/include")
target_link_libraries(nt_ui PRIVATE lvgl m)
